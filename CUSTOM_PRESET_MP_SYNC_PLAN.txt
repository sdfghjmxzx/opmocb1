CUSTOM CHARACTER PRESET MULTIPLAYER SYNCHRONIZATION
IMPLEMENTATION PLAN
====================================================

PROBLEM:
--------
When a player selects a custom character preset in multiplayer lobby, the opponent 
only receives the character name. The opponent's client searches their local 
characters.json for that name, but custom presets only exist locally, resulting in 
None preset and default stats (50 across all attributes).

SOLUTION:
---------
Implement delta-based synchronization of custom character data through server relay,
with client-side temp preset creation for opponent visualization.

ARCHITECTURE OVERVIEW:
----------------------

1. CLIENT SEND (Player with custom preset)
   - Detects custom/temp preset selection
   - Compares with last sent data (delta tracking)
   - Sends only changed fields to server
   - Stores sent data for future comparison

2. SERVER RELAY
   - Receives character selection with optional custom_data/deltas
   - Merges deltas into stored full character_data per player
   - Broadcasts full merged character_data to all lobby players

3. CLIENT RECEIVE (Opponent)
   - Receives lobby_state_update with character_data
   - Creates/updates temp preset "OPPONENT_CUSTOM" in main_menu_mp_temp_presets
   - Updates only changed fields (if deltas provided)
   - Preset automatically appears in opponent's gallery

4. LIFECYCLE MANAGEMENT
   - CREATE: First custom selection by opponent
   - UPDATE: Each subsequent custom selection (deltas only)
   - DELETE: Opponent leaves, you leave, or opponent switches to non-custom

def send_mp_character_select(char_name, preset_data=None):
    """Send character selection to server with optional custom data."""
    if websockets is not None and network_client is not None:
        network_client.start()
        
        # Check if this is a custom/temp preset
        is_custom = False
        custom_data = None
        deltas = None
        
        if preset_data:
            is_custom = preset_data.get("is_custom", False) or preset_data.get("is_temp", False)
        
        if is_custom and preset_data:
            # Extract current preset data
            current_data = {
                "name": preset_data.get("name"),
                "stats": preset_data.get("stats", {}),
                "power": preset_data.get("power")
            }
            
            # Compare with last sent data to find deltas
            global main_menu_mp_last_sent_custom
            last_sent = main_menu_mp_last_sent_custom
            
            deltas = {}
            
            # Check name change
            if current_data["name"] != last_sent.get("name"):
                deltas["name"] = current_data["name"]
            
            # Check power change
            if current_data["power"] != last_sent.get("power"):
                deltas["power"] = current_data["power"]
            
            # Check stat changes
            current_stats = current_data["stats"]
            last_stats = last_sent.get("stats", {})
            
            for stat_key in ["strength", "defense", "speed", "reaction", "endurance", "willpower", "haki", "devil_fruit"]:
                current_val = current_stats.get(stat_key, 50)
                last_val = last_stats.get(stat_key, 50)
                if current_val != last_val:
                    if "stats" not in deltas:
                        deltas["stats"] = {}
                    deltas["stats"][stat_key] = current_val
            
            # If this is first send or substantial changes, send full data
            if last_sent.get("name") is None or len(deltas) > 0:
                custom_data = current_data if last_sent.get("name") is None else deltas
                
                # Update last sent tracker
                store.main_menu_mp_last_sent_custom = current_data
        
        # Send to server
        network_client.send_character_selection(char_name, custom_data)
    
    poll_network_messages()
    renpy.restart_interaction()
```

# Handle opponent custom character temp preset creation
global main_menu_mp_temp_presets, main_menu_mp_session_id

my_session = main_menu_mp_session_id

# Process each player's character data
for sid, pdata in players.items():
    if sid == my_session:
        # Skip own data
        continue
    
    char_name = pdata.get('selected_character', 'None')
    char_data = pdata.get('character_data', None)
    
    if char_data:
        # Opponent has custom character - create/update temp preset
        opponent_preset_name = "OPPONENT_CUSTOM"
        
        # Find existing opponent temp preset
        existing_preset = None
        for preset in main_menu_mp_temp_presets:
            if preset.get("name") == opponent_preset_name:
                existing_preset = preset
                break
        
        if existing_preset:
            # Update existing preset
            if "name" in char_data:
                # Name changed - this is unusual but handle it
                pass  # Keep OPPONENT_CUSTOM as display name
            
            if "power" in char_data:
                existing_preset["power"] = char_data["power"]
            
            if "stats" in char_data:
                if "stats" not in existing_preset:
                    existing_preset["stats"] = {}
                # Merge stats
                for stat_key, stat_val in char_data["stats"].items():
                    existing_preset["stats"][stat_key] = stat_val
            
            print(f"[CLIENT] Updated OPPONENT_CUSTOM temp preset")
        else:
            # Create new temp preset
            new_preset = {
                "name": opponent_preset_name,
                "picture": "images/characters/unknown.png",
                "stats": char_data.get("stats", {
                    "strength": 50,
                    "defense": 50,
                    "speed": 50,
                    "reaction": 50,
                    "endurance": 50,
                    "willpower": 50,
                    "haki": 50,
                    "devil_fruit": 50
                }),
                "power": char_data.get("power", "None"),
                "is_temp": True
            }
            main_menu_mp_temp_presets.append(new_preset)
            print(f"[CLIENT] Created OPPONENT_CUSTOM temp preset")
        
        # Update selection variable to point to temp preset
        if sid == host_session:
            store.main_menu_mp_p1_selected = opponent_preset_name
        else:
            store.main_menu_mp_p2_selected = opponent_preset_name
    
    else:
        # Opponent switched to non-custom character
        # Remove OPPONENT_CUSTOM temp preset if exists
        opponent_preset_name = "OPPONENT_CUSTOM"
        store.main_menu_mp_temp_presets = [
            p for p in main_menu_mp_temp_presets 
            if p.get("name") != opponent_preset_name
        ]
```
