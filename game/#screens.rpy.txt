    def get_player_zoom(player):
        if player.hopping:
            import time, math
            elapsed = time.time() - player.hop_start_time
            hop_duration = player.hop_duration
            current_hop = int(elapsed / hop_duration)
            
            # If we've completed all hops, stop hopping
            if current_hop >= player.hop_count:
                player.hopping = False
                return player.hop_base_zoom
            
            # Calculate progress within current hop (0 to 1)
            hop_progress = (elapsed % hop_duration) / hop_duration
            
            # Create a bounce effect - zoom in at the peak of the hop
            # Use a sine wave for smooth bouncing
            bounce_factor = math.sin(hop_progress * math.pi)
            zoom_factor = 1.0 + (bounce_factor * 0.4)  # 40% zoom at peak
            
            return player.hop_base_zoom * zoom_factor
        else:
            return player.hop_base_zoom

        def get_current_player(self):
            if self.phase == "attack":
                return self.player1 if self.attacker_is_p1 else self.player2
            return self.player2 if self.attacker_is_p1 else self.player1

        def get_opponent(self):
            cp = self.get_current_player()
            return self.player2 if cp == self.player1 else self.player1

        # --- Planning ---
        def start_movement_planning(self):
            if self.planning_mode:
                return
            self.planning_mode = True
            self.movement_mode = True
            p = self.get_current_player()
            self.ghost_row, self.ghost_col = p.row, p.col
            self.ghost_facing = p.facing
            self.current_path = [(p.row, p.col)]
            self.planned_actions = []
            self.planning_terminal = False
            self._recompute_highlights()

        def _neighbors(self, r, c):
            nbrs = []
            for dr in (-1, 0, 1):
                for dc in (-1, 0, 1):
                    if dr == 0 and dc == 0:
                        continue
                    nr = r + dr
                    nc = c + dc
                    if 0 <= nr < 7 and 0 <= nc < 7:
                        nbrs.append((nr, nc))
            return nbrs

        def _recompute_highlights(self):
            if self.ghost_row is None:
                self.highlighted_squares = []
                self.attack_highlighted_squares = []
                return
            # Movement range (yellow)
            self.highlighted_squares = self._neighbors(self.ghost_row, self.ghost_col)
            # Attack preview (red) – minimal slice: single tile ahead based on ghost_facing
            self.attack_highlighted_squares = self._compute_attack_pattern_preview()

        def _compute_attack_pattern_preview(self):
            tiles = []
            if self.ghost_row is None:
                return tiles
            r = self.ghost_row
            c = self.ghost_col
            ang = int(self.ghost_facing) % 360
            if ang == 90 and r + 1 < 7:
                tiles = [(r + 1, c)]
            elif ang == 270 and r - 1 >= 0:
                tiles = [(r - 1, c)]
            elif ang == 0 and c - 1 >= 0:
                tiles = [(r, c - 1)]
            elif ang == 180 and c + 1 < 7:
                tiles = [(r, c + 1)]
            return tiles

        def add_to_path(self, row, col):
            if not self.movement_mode or self.planning_terminal:
                raise PlanningTerminalError("Cannot move after selecting attack/defense")
            # Chebyshev-1 step
            if (row, col) in self._neighbors(*self.current_path[-1]):
                self.current_path.append((row, col))
                self.ghost_row, self.ghost_col = row, col
                self.planned_actions.append(("move", {"tile": f"{row+1}{chr(65+col)}"}))
                self._recompute_highlights()

        def remove_last_step(self):
            if not self.movement_mode or len(self.current_path) <= 1:
                return
            self.current_path.pop()
            self.ghost_row, self.ghost_col = self.current_path[-1]
            # Remove last move action
            for i in range(len(self.planned_actions) - 1, -1, -1):
                if self.planned_actions[i][0] == "move":
                    self.planned_actions.pop(i)
                    break
            self._recompute_highlights()

        def add_rotation(self, degrees):
            if self.planning_terminal:
                raise PlanningTerminalError("Cannot rotate after selecting attack/defense")
            if not self.planning_mode:
                return
            self.ghost_facing = (self.ghost_facing + degrees) % 360
            self.planned_actions.append(("rotate", degrees))
            self._recompute_highlights()

        def add_attack(self, kind):
            if not self.planning_mode:
                return
            self.planning_terminal = True
            self.movement_mode = False
            self.planned_actions.append(("attack", kind))
            self._recompute_highlights()

        def add_defense(self, kind):
            if not self.planning_mode:
                return
            self.planning_terminal = True
            self.movement_mode = False
            self.planned_actions.append(("defense", kind))
            self._recompute_highlights()

        def undo_last_planned_action(self):
            if not self.planned_actions:
                return
            last = self.planned_actions.pop()
            if last[0] in ("attack", "defense"):
                self.planning_terminal = False
                self.movement_mode = True
            self._recompute_highlights()

        def confirm_turn(self):
            if not self.planning_mode:
                return
            p = self.get_current_player()
            # Apply movement
            if self.current_path:
                p.row, p.col = self.current_path[-1]
                p.facing = int(self.ghost_facing) % 360
            if self.phase == "attack":
                skip = any(a for a in self.planned_actions if a[0] == "attack" and a[1] == "skip")
                attack_selected = any(a for a in self.planned_actions if a[0] == "attack" and a[1] != "skip")
                miss = False
                if attack_selected:
                    attack_tiles = self._compute_attack_pattern_preview()
                    d = self.get_opponent()
                    miss = (d.row, d.col) not in attack_tiles
                if skip or miss:
                    # Skip defender phase; defender becomes next attacker
                    self.phase = "attack"
                    self.attacker_is_p1 = not self.attacker_is_p1
                else:
                    self.phase = "defense"
            else:
                # Defense resolved; next attacker is prior defender
                self.phase = "attack"
                self.attacker_is_p1 = not self.attacker_is_p1
            self.cancel_planning()

        def cancel_planning(self):
            self.planning_mode = False
            self.movement_mode = False
            self.planning_terminal = False
            self.current_path = []
            self.highlighted_squares = []
            self.attack_highlighted_squares = []
            self.ghost_row = None
            self.ghost_col = None
            self.ghost_facing = None

        # --- Wheel controller ---
        def get_wheel_rotation(self):
            return self.wheel_rotation

        def start_wheel_drag(self, initial_mouse_angle):
            self.wheel_dragging = True
            self.wheel_drag_initial_mouse_angle = initial_mouse_angle
            if self.ghost_facing is None:
                self.ghost_facing = self.get_current_player().facing
            self.wheel_drag_start_facing = self.ghost_facing

        def update_wheel_drag(self, new_wheel_angle):
            self.wheel_rotation = new_wheel_angle
            # Directly control ghost orientation during drag
            self.ghost_facing = new_wheel_angle % 360
            self._recompute_highlights()

        def end_wheel_drag(self):
            self.wheel_dragging = False

        # --- Costs ---
        def get_movement_cost(self, steps):
            base = 15 if self.phase == "attack" else 30
            return base * max(0, steps)

        @property
        def total_cost(self):
            steps = max(0, len(self.current_path) - 1)
            return self.get_movement_cost(steps)

        def get_player_pos(player):
            if player.animating and player.animation_path:
                import time
                if not hasattr(player, 'anim_start_time'):
                    player.anim_start_time = time.time()
                
                elapsed = time.time() - player.anim_start_time
                step_duration = 0.3  # seconds per step
                current_step = min(int(elapsed / step_duration), len(player.animation_path) - 1)
                
                if current_step >= len(player.animation_path) - 1:
                    # Animation complete - UPDATE ACTUAL POSITION NOW
                    player.animating = False
                    target_row, target_col = player.animation_path[-1]
                    player.row = target_row
                    player.col = target_col
                    del player.anim_start_time
                    combat_game.start_next_animation(player)
                    return (player.row, player.col)
                
                # Interpolate between steps
                step_progress = (elapsed % step_duration) / step_duration
                from_row, from_col = player.animation_path[current_step]
                to_row, to_col = player.animation_path[current_step + 1]
                
                anim_row = from_row + (to_row - from_row) * step_progress
                anim_col = from_col + (to_col - from_col) * step_progress
                
                return (anim_row, anim_col)
            else:
                return (player.row, player.col)
        
        def get_player_facing(player):
            if player.rotating:
                import time
                elapsed = time.time() - player.rotation_start_time
                rotation_duration = 0.5  # seconds for rotation
                
                if elapsed >= rotation_duration:
                    player.rotating = False
                    # UPDATE ACTUAL FACING NOW
                    player.facing = player.target_facing
                    # Start next animation
                    combat_game.start_next_animation(player)
                    return player.facing
                
                # Smooth rotation interpolation
                progress = elapsed / rotation_duration
                if not hasattr(player, 'rotation_old_facing'):
                    return player.facing
                
                old_facing = player.rotation_old_facing
                new_facing = player.target_facing
                
                # Handle 360-degree wrapping - take shortest path
                diff = new_facing - old_facing
                if diff > 180:
                    diff -= 360
                elif diff < -180:
                    diff += 360
                
                # Interpolate
                current_facing = old_facing + (diff * progress)
                return current_facing % 360
            else:
                return player.facing

# Wheel interaction system
init python:
    def check_wheel_drag_state(wheel_x, wheel_y):
        """Check if mouse button is pressed and start/continue drag"""
        # Check if left mouse button is pressed
        mouse_pressed = renpy.get_mouse_pos() and pygame.mouse.get_pressed()[0]
        
        if mouse_pressed:
            if not combat_game.wheel_dragging:
                # Start new drag - save initial mouse angle
                mouse_x, mouse_y = renpy.get_mouse_pos()
                dx = mouse_x - wheel_x
                dy = mouse_y - wheel_y
                initial_mouse_angle = math.degrees(math.atan2(dy, dx))
                if initial_mouse_angle < 0:
                    initial_mouse_angle += 360
                combat_game.start_wheel_drag(initial_mouse_angle)
            # Update drag position
            update_wheel_from_mouse(wheel_x, wheel_y)
            renpy.restart_interaction()
    
    def update_wheel_from_mouse(wheel_x, wheel_y):
        """Update wheel rotation based on mouse position - RELATIVE rotation"""
        if not combat_game.wheel_dragging:
            return
                
        # Get current mouse position
        mouse_x, mouse_y = renpy.get_mouse_pos()
        
        # Calculate angle from wheel center to mouse
        dx = mouse_x - wheel_x
        dy = mouse_y - wheel_y
        
        # Avoid division by zero
        if dx == 0 and dy == 0:
            return
                
        # Calculate current mouse angle
        current_mouse_angle = math.degrees(math.atan2(dy, dx))
        if current_mouse_angle < 0:
            current_mouse_angle += 360
        
        # Calculate how much the mouse has rotated since drag started
        angle_delta = current_mouse_angle - combat_game.wheel_drag_initial_mouse_angle
        
        # Handle wraparound
        if angle_delta > 180:
            angle_delta -= 360
        elif angle_delta < -180:
            angle_delta += 360
        
        # Apply the rotation delta to the starting wheel facing
        new_wheel_angle = (combat_game.wheel_drag_start_facing + angle_delta) % 360
                
        combat_game.update_wheel_drag(new_wheel_angle)
